CGame::bGetItemNameWhenDeleteNpc (int &iItemID, short sNpcType)
{
xmlDoc *doc = NULL;
xmlNode *root_element = NULL;
doc = xmlParseFile("NpcItem.xml");
if (doc == NULL)
{
 PutLogList("error: could not parse file file.xml\n");
}
root_element = xmlDocGetRootElement(doc);
root_element=root_element->children;
xmlNode *cur_node = NULL;
xmlNode *sub_node = NULL;
xmlNode *drop_node = NULL;
xmlChar *DiceMin = NULL;
xmlChar *DiceMax = NULL;
xmlChar *NpcID= NULL;
xmlChar *RollMin=NULL;
xmlChar *RollMax=NULL;
xmlChar *ItemID=NULL;
char stype[10];
bool bNpc;
int Roll;
char ctxt[100],ctmp[100];
bNpc=FALSE;
iItemID=-1;
wsprintf(stype,"%d",sNpcType);
for (cur_node = root_element; cur_node; cur_node = cur_node->next)
{
 if (xmlStrcmp(cur_node->name, (const xmlChar *)"NPC") == 0)
 {
  DiceMin = xmlGetProp(cur_node, (const xmlChar *)"DiceMin");
  DiceMax = xmlGetProp(cur_node, (const xmlChar *)"DiceMax");
  NpcID = xmlGetProp(cur_node, (const xmlChar *)"ID");
  if ((xmlStrcmp(NpcID, (const xmlChar *)stype) == 0)&&(!bNpc))
  {
   bNpc=TRUE;
   Roll=iDice(atoi((const char *)DiceMin),atoi((const char *)DiceMax));
   for (sub_node = cur_node->children; sub_node; sub_node = sub_node->next)
   {
    if (xmlStrcmp(sub_node->name,(const xmlChar *)"DROP")==0)
    {
     RollMin=xmlGetProp(sub_node,(const xmlChar *)"RollMin");
     RollMax=xmlGetProp(sub_node,(const xmlChar *)"RollMax");
     ItemID=xmlGetProp(sub_node,(const xmlChar *)"ItemID");
     if ( (atoi((const char *)RollMin)<Roll) && (atoi((const char *)RollMax)>Roll) && (iItemID == -1))
     {
      iItemID=atoi((const char *)ItemID);
     }
    }
   }
  }
 }
}
xmlFreeDoc(doc);
xmlCleanupParser();
return iItemID != -1;
}
