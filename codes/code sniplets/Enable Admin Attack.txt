void CGame::AdminOrder_EnableAdminAttack(int iClientH, char *pData, DWORD dwMsgSize)
{
char   * token, cBuff[256], len;
char   seps[] = "= \t\n";
class  CStrTok * pStrTok;

if (m_pClientList[iClientH] == NULL) return;
if ((dwMsgSize) <= 0) return;

ZeroMemory(cBuff, sizeof(cBuff));
memcpy(cBuff, pData, dwMsgSize);
pStrTok = new class CStrTok(cBuff, seps);
token = pStrTok->pGet();
token = pStrTok->pGet();

if (token != NULL) 
{
 len = strlen(token);
 if(len > 10) len = 10;
 if (memcmp(token, m_cSecurityNumber, len) == 0) 
 {
  if (m_pClientList[iClientH]->m_bAdminAttackEnabled == FALSE) m_pClientList[iClientH]->m_bAdminAttackEnabled = TRUE;
  else m_pClientList[iClientH]->m_bAdminAttackEnabled = FALSE;
 }
}
}
