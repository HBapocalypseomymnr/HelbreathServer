void CGame::PlayerCommand_RepReset(int iClientH)
{
char cGizonLow[256], cGizonDone[256];

if (m_pClientList[iClientH] == NULL) return;
if (m_pClientList[iClientH]->m_iGizonItemUpgradeLeft < 5){
ZeroMemory(cGizonLow, sizeof(cGizonLow));
wsprintf(cGizonLow,"You Need 5 Majestic points");
ShowClientMsg(iClientH, cGizonLow);
}
else{
m_pClientList[iClientH]->m_iRating = 0; 
ZeroMemory(cGizonDone, sizeof(cGizonDone));
wsprintf(cGizonDone, "Your Rep Is Now 0!");
ShowClientMsg(iClientH, cGizonDone);
m_pClientList[iClientH]->m_iGizonItemUpgradeLeft -= 5;
DeleteClient(iClientH, TRUE, TRUE);
}
}
