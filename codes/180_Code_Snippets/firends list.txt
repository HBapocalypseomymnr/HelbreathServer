void CGame::SaveFriendList()
{
FILE* f=fopen("contents\\FriendList.txt","w");
int i;
char wr[12];
for (i=0;i<m_iTotalFriends;i++) {
ZeroMemory(wr,sizeof(wr));
wsprintf(wr,"%s\n",m_cFriends[i]);
fputs(wr,f);
}
fclose(f);
}

{
 char buf[1024];
 int konieclinii;

   konieclinii=0;
   int i=strlen(buf);
